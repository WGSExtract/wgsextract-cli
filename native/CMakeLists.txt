cmake_minimum_required(VERSION 3.24)
set(CMAKE_OSX_DEPLOYMENT_TARGET 13.0 CACHE STRING "Minimum macOS version")
project(wgse_native_aligners VERSION 0.1.0 LANGUAGES C CXX)

if(APPLE AND CMAKE_SYSTEM_PROCESSOR MATCHES "^(arm64|aarch64)$")
    enable_language(OBJCXX)
    set(WGSE_METAL ON)
elseif(CMAKE_SYSTEM_NAME STREQUAL "Linux" AND CMAKE_SYSTEM_PROCESSOR MATCHES "^(x86_64|AMD64)$")
    # CPU-only builds (#671).
    set(WGSE_METAL OFF)
else()
    message(FATAL_ERROR "The wgse native aligners build on Apple Silicon macOS and x86-64 Linux")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_OBJCXX_STANDARD 17)
set(CMAKE_POLICY_VERSION_MINIMUM 3.5)
include(FetchContent)
include(CTest)

if(WGSE_METAL)
    add_subdirectory(metal-common)
    add_subdirectory(bwa-mem2)
    add_subdirectory(bwa)
    add_subdirectory(minimap2)
    add_subdirectory(bwa-mem3)
else()
    add_subdirectory(bwa)
endif()
